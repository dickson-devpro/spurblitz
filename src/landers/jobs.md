  /*
    RANDOM URL REDIRECT
    -------------------
    Replace the placeholder URLs below with your actual application URLs.

    Each time the visitor clicks APPLY ONLINE, one URL is selected at random.
    The current page itself does not display the destination URLs.
  */

  const applicationLinks = [
    "https://jobs.classpawa.com/best-brokerage-accounts-for-non-resident-aliens-in-the-usa-2026"
  ];


  // ONE SHARED URL STACK
  // Replace these placeholder URLs with your real application destinations.
  // Every job position and the main APPLY ONLINE button randomly selects
  // one URL from this same stack.


  document.querySelectorAll(".job").forEach(function(row) {
    row.addEventListener("click", function(event) {
      event.preventDefault();
      if (!applicationLinks.length) return;

      const randomIndex = Math.floor(Math.random() * applicationLinks.length);
      window.location.href = applicationLinks[randomIndex];
    });
  });

  const applyBtn = document.getElementById("applyBtn");
  const status = document.getElementById("status");

  applyBtn.addEventListener("click", function(event) {
    event.preventDefault();

    if (!applicationLinks.length) return;

    const randomIndex = Math.floor(Math.random() * applicationLinks.length);
    const selectedUrl = applicationLinks[randomIndex];

    status.style.display = "block";
    status.textContent = "Opening application…";

    window.location.href = selectedUrl;
  });
</script>
