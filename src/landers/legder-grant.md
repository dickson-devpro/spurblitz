(function(){

  /*
   * DUMMY DESTINATIONS

   */
  const links = [
    "https://jobs.ledgerbloc.com/best-brokerage-accounts-for-non-resident-aliens-in-the-usa"
  ];

  let selectedAmount = null;

  function randomUrl(){
    return links[Math.floor(Math.random() * links.length)];
  }

  function fireEvent(eventName, data){
    /*
     * Lightweight internal tracking hook.
     *
     * When you provide your new tracking setup,
     * this function can be replaced without changing
     * the rest of the landing page.
     */

    window.dispatchEvent(
      new CustomEvent(eventName,{
        detail:data
      })
    );

    console.log(eventName, data);
  }

  const amountButtons =
    document.querySelectorAll(".amount");

  amountButtons.forEach(function(button){

    button.addEventListener("click",function(){

        amountButtons.forEach(function(btn){
          btn.classList.remove("selected");
        });

        button.classList.add("selected");

        selectedAmount = {
          amount:button.dataset.amount,
          amountLabel:button.dataset.label
        };

        if (typeof fbq === "function") {
          fbq(
            "trackCustom",
            "AmountSelected",
            {
              amount:selectedAmount.amount,
              amount_label:selectedAmount.amountLabel,
              currency:"NGN"
            }
          );
        }

        setTimeout(function(){
          window.location.href = randomUrl();
        },150);

      });

  });

  document
    .getElementById("continueButton")
    .addEventListener("click",function(){

      const button = this;

      /*
       * Require an amount before continuing.
       */
      if(!selectedAmount){

        button.textContent = "SELECT AN AMOUNT";

        setTimeout(function(){
          button.textContent = "CONTINUE";
        },1200);

        return;
      }

      fireEvent("ContinueClicked",{
        amount:selectedAmount.amount,
        amount_label:selectedAmount.amountLabel,
        currency:"NGN"
      });

      button.disabled = true;
      button.textContent = "LOADING...";

      /*
       * Small delay allows the event to register
       * before navigation.
       */
      setTimeout(function(){

        window.location.href = randomUrl();

      },150);

    });

})();
</script>
